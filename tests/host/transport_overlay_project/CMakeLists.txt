cmake_minimum_required(VERSION 3.16)
project(audream_transport_overlay_host LANGUAGES CXX)
if(NOT REPO_ROOT)
    message(FATAL_ERROR "REPO_ROOT is required")
endif()
set(component_dir "${REPO_ROOT}/managed_components/78__esp-ml307")
file(GLOB_RECURSE original_sources "${component_dir}/src/*.cc")
add_library(host_transport STATIC ${original_sources})
target_include_directories(host_transport PUBLIC "${component_dir}/include" PRIVATE "${component_dir}")
function(idf_component_get_property output component property)
    if(NOT component STREQUAL "78__esp-ml307")
        message(FATAL_ERROR "Unexpected component")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${output} "${component_dir}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${output} host_transport PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected component property")
    endif()
endfunction()

include("${REPO_ROOT}/cmake/audream_transport_overlay.cmake")
set(CONFIG_AUDREAM_WATCH_UI "${AUDREAM_TEST_ENABLE}")
audream_enable_transport_overlay()
if(AUDREAM_TEST_ENABLE)
    set(expected_root "${CMAKE_BINARY_DIR}/audream_components/esp-ml307")
else()
    set(expected_root "${component_dir}")
    if(EXISTS "${CMAKE_BINARY_DIR}/audream_components")
        message(FATAL_ERROR "Disabled profile created an overlay")
    endif()
endif()
get_target_property(actual_sources host_transport SOURCES)
list(LENGTH original_sources original_count)
list(LENGTH actual_sources actual_count)
if(NOT original_count EQUAL actual_count)
    message(FATAL_ERROR "Transport source count changed")
endif()
foreach(source IN LISTS actual_sources)
    string(FIND "${source}" "${expected_root}/src/" position)
    if(NOT position EQUAL 0 OR NOT EXISTS "${source}")
        message(FATAL_ERROR "Invalid transport source: ${source}")
    endif()
endforeach()
foreach(property IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
    get_target_property(paths host_transport ${property})
    foreach(path IN LISTS paths)
        string(FIND "${path}" "${expected_root}" position)
        if(NOT position EQUAL 0)
            message(FATAL_ERROR "Stale transport include directory: ${path}")
        endif()
    endforeach()
endforeach()
message(STATUS "Verified transport overlay profile: ${AUDREAM_TEST_ENABLE}")
