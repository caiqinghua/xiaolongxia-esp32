function(idf_component_get_property output component property)
    if(NOT component STREQUAL "bt")
        message(FATAL_ERROR "Unexpected component")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${output} "${component_dir}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${output} host_bt PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected property")
    endif()
endfunction()

    function(extract_bta_function source start_marker end_marker output)
        file(READ "${source}" text)
        string(FIND "${text}" "${start_marker}" start)
        string(FIND "${text}" "${end_marker}" end)
        if(start LESS 0 OR end LESS start)
            message(FATAL_ERROR "Reviewed BTA function boundaries are missing")
        endif()
        math(EXPR length "${end} - ${start}")
        string(SUBSTRING "${text}" ${start} ${length} body)
        file(WRITE "${CMAKE_BINARY_DIR}/${output}" "${body}\n")
    endfunction()
