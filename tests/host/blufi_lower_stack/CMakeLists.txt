project(audream_blufi_lower_stack_host LANGUAGES C)
if(NOT IDF_BT_DIR OR NOT REPO_ROOT)
    message(FATAL_ERROR "IDF_BT_DIR and REPO_ROOT are required")
endif()
set(stack "${IDF_BT_DIR}/host/bluedroid/stack/l2cap")
foreach(entry IN ITEMS
    "l2c_utils.c|5b4896647bdc5963d2e01dddbcd175c02056bd14b60c78c6591b4f4713ccc8ec"
    "l2c_link.c|ae7c6eed4d84c56bf79e3948ccabbb822d9eff294ff6564e6762aaf355ae3cf0")
    string(REPLACE "|" ";" fields "${entry}")
    list(GET fields 0 name)
    list(GET fields 1 expected)
    file(SHA256 "${stack}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Unreviewed lower-stack source: ${name}")
    endif()
endforeach()

function(extract_function source signature output)
    file(READ "${source}" text)
    string(FIND "${text}" "${signature}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing reviewed function: ${signature}")
    endif()
    string(SUBSTRING "${text}" ${start} -1 body)
    # These hash-checked functions use a column-zero closing brace only at end.
    string(FIND "${body}" "\n}\n" end)
    if(end LESS 0)
        message(FATAL_ERROR "Missing reviewed function end: ${signature}")
    endif()
    math(EXPR length "${end} + 3")
    string(SUBSTRING "${body}" 0 ${length} body)
    file(WRITE "${CMAKE_BINARY_DIR}/${output}" "${body}\n")
endfunction()
