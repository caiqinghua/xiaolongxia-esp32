cmake_minimum_required(VERSION 3.20)
project(watch_motion_overlay_host LANGUAGES CXX)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT DEFINED SENSORLIB)
    message(FATAL_ERROR "SENSORLIB is required")
endif()
set(CONFIG_AUDREAM_WATCH_UI TRUE)
function(idf_build_get_property output property)
    if(NOT property STREQUAL "BUILD_COMPONENTS")
        message(FATAL_ERROR "Unexpected build property")
    endif()
    set(${output} lewisxhe__sensorlib PARENT_SCOPE)
endfunction()
function(idf_component_get_property output component property)
    if(NOT component STREQUAL "lewisxhe__sensorlib")
        message(FATAL_ERROR "Unexpected component identity")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${output} "${SENSORLIB}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${output} sensor_fixture PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected component property")
    endif()
endfunction()
add_library(sensor_fixture STATIC "${SENSORLIB}/src/sensor/imu/qmi8658/SensorQMI8658.cpp")
target_include_directories(sensor_fixture PUBLIC "${SENSORLIB}/src")
include("${root}/cmake/audream_motion_overlay.cmake")
audream_enable_motion_overlay()
foreach(property IN ITEMS SOURCES INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
    get_target_property(values sensor_fixture ${property})
    foreach(value IN LISTS values)
        if(NOT value MATCHES "^${CMAKE_BINARY_DIR}/audream_components/sensorlib/src")
            message(FATAL_ERROR "Unpatched source or include remains in ${property}: ${value}")
        endif()
    endforeach()
endforeach()
get_target_property(definitions sensor_fixture INTERFACE_COMPILE_DEFINITIONS)
if(NOT "AUDREAM_SENSORLIB_LIFECYCLE_PATCHED=1" IN_LIST definitions)
    message(FATAL_ERROR "The patch requirement did not propagate to consumers")
endif()
message(STATUS "Motion target overlay: sources, headers, and required patch marker passed")
